# Skene example
#
# The smallest run that draws: a place, a store, two workers and a reader. It carries one chain,
# a fact, a declaration that starts from it, the worker that took it, and the fact it produced.
#
#   ktema-sim scenario/example.txt /tmp/example-volume --reset

seed 7

tick 0
position blackboard 260,88
position worker-1 110,56
position worker-2 110,120
position reader-1 410,88
count 4

# The reading the work starts from.
fact reading-1 48 worker-1/observation
count 5

# What bounded the judgement.
hint bound ceiling=one,floor=zero

# The declaration, the taker and the outcome.
intent answer reading-1 take-the-work
claim answer worker-1
fact answered 48 worker-1/result
count 6
conclude answer work-done
count 7

flush

# A second worker reads the same record and takes nothing.
tick 3600
fact reading-2 48 worker-2/observation
count 8

query 0 15540 * 8
